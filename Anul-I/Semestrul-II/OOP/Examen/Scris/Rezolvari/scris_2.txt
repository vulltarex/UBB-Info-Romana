class Meniu {
private:
    int pret;
public:
    Meniu(int pret) : pret(pret) {}
    virtual int getPret() { return pret; }
    virtual std::string descriere() = 0;
    virtual ~Meniu() = default;
};

class CuCafea : public Meniu {
private:
    Meniu* meniu;
public:
    CuCafea(Meniu* m) : Meniu(m->getPret()), meniu(m) {}
    int getPret() override { return meniu->getPret() + 5; }
    std::string descriere() override { return meniu->descriere() + " cu cafea"; }
    ~CuCafea() { delete meniu; }
};

class CuRacoritoare : public Meniu {
private:
    Meniu* meniu;
public:
    CuRacoritoare(Meniu* m) : Meniu(m->getPret()), meniu(m) {}
    int getPret() override { return meniu->getPret() + 4; }
    std::string descriere() override { return meniu->descriere() + " cu racoritoare"; }
    ~CuRacoritoare() { delete meniu; }
};

class MicDejun : public Meniu {
private:
    std::string denumire;
public:
    MicDejun(const std::string& denumire, int pret) : Meniu(pret), denumire(denumire) {}
    std::string descriere() override { return denumire; }
};

//3.2
std::vector<Meniu*> listaMeniuri() {
    std::vector<Meniu*> rez;
    rez.push_back(new CuCafea(new CuRacoritoare(new MicDejun("Omleta", 15))));
    rez.push_back(new CuCafea(new MicDejun("Ochiuri", 10)));
    rez.push_back(new MicDejun("Omleta", 15));
    return rez;
}

//3.3
int main() {
    std::vector<Meniu*> rez = listaMeniuri();
    std::sort(rez.begin(), rez.end(), [](const auto a, const auto b) { return a->getPret() > b->getPret(); });
    for (const auto m : rez) {
        std::cout << m->descriere() << ' ' << m->getPret() << " RON\n";
        delete m;
    }
    return 0;
}

template <typename tip>
class Measurement {
public:
    tip value;
    Measurement(tip val) : value(val) {}
    Measurement& operator+(tip val) {
        return Measurement(this->value + val);
    }
    bool operator<=(const Measurement& other) {
        return this->value <= other.value;
    }
    bool operator<(const Measurement& other) {
        return this->value < other.value;
    }
};
