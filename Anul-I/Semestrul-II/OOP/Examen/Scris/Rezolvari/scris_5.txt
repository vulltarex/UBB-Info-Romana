class Participant {
public:
    Participant() = default;
    virtual void tipareste() = 0;
    virtual bool eVoluntar() { return true; }
    virtual ~Participant() = default;
};

class Angajat : public Participant {
private:
    Participant* participant;
public:
    Angajat(Participant* p) : participant(p) {}
    void tipareste() override {
        participant->tipareste();
        std::cout << " angajat";
    }
    bool eVoluntar() override { return false; }
    ~Angajat() { delete participant; }
};

class Director : public Personal {
public:
    Director(std::string& nume) : Personal(nume) {}
    void tipareste() override {
        Personal::tipareste();
        std::cout << " Director";
    }
};

class ONG {
private:
    std::vector<Participant*> participanti;
public:
    ONG() = default;
    void add(Participant* p) {
        participanti.push_back(p);
    }
    std::vector<Participant*> getAll(bool voluntari) const {
        std::vector<Participant*> rez;
        for (auto p : participanti) {
            if (p->eVoluntar() == voluntari) {
                rez.push_back(p);
            }
        }
        return rez;
    }
    ~ONG() {
        for (auto p : participanti) delete p;
    }
};

ONG func() {

int main() {
    ONG ong = func();

    for (auto p : voluntari) {
        p->tipareste();
        std::cout << '\n';
    }
    return 0;
}

template <typename T>
class Cos {
private:
    std::vector<Cos> stari;
    std::vector<T> cos;
public:
    Cos(const Cos& other) {
        for (auto stare : other.stari)
            this->stari.push_back(stare);
        for (auto obiect : other.cos)
            this->cos.push_back(obiect);
    }
    Cos& operator+(T& e) {
        stari.push_back(*this);
        cos.push_back(e);
        return *this;
    }
    void undo() {
        *this = stari.pop_back();
    }
    ~Cos() = default;
};
