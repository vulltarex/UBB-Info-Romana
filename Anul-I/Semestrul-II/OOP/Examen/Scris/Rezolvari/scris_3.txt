class Transformer {
public:
    Transformer() = default;
    virtual void transform(std::vector<int>& nrs) = 0;
    virtual ~Transformer() = default;
};

class Composite : public Transformer {
private:
    Transformer* t1;
    Transformer* t2;
public:
    Composite(Transformer* tr1, Transformer* tr2) : t1(tr1), t2(tr2) {}
    void transform(std::vector<int>& nrs) override {
        t1->transform(nrs);
        t2->transform(nrs);
    }
    ~Composite() {
        delete t1;
        delete t2;
    }
};

class Nuller : public Adder {
public:
    Nuller(int cat) : Adder(cat) {}
    void transform(std::vector<int>& nrs) override {
        std::vector<int> rez;
        for (int nr : nrs) rez.push_back(nr);
        Adder::transform(rez);
        for (int i = 0; i < nrs.size(); i++) {
            if (rez[i] > 10) nrs[i] = 0;
        }
    }
};

template <typename T>
class ToDo {
private:
    std::vector<T> toDoList;
public:
    void printToDoList(std::ostream& s) {
        s << "De facut:";
        for (const T& task : toDoList) {
            s << task << ';';
        }
    }
    ToDo& operator<<(const T& task) {
        toDoList.push_back(task);
        return *this;
    }
};
